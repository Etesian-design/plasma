# The Etesian mail plugin (QML module org.etesian.mail): the mails behind the Belle Mail widget (org.etesian.mail plasmoid).
# Sources: Thunderbird (the profile's inboxes) and KMail (Akonadi). Built by ./build.sh (here or in a Kubuntu container); installed by ../../install.sh into
# ~/.local/lib/qt6/qml/org/etesian/mail.
cmake_minimum_required(VERSION 3.22)
project(etesian-mail LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Qml DBus Sql)
# Thunderbird (its profile's mbox inboxes) needs only Qt; KMail needs Akonadi (KDE PIM) and is left out without it
set(SOURCES plugin.cpp thunderbirdinbox.cpp thunderbirdinbox.h thunderbirdsearch.cpp thunderbirdsearch.h)
set(LIBS Qt6::Core Qt6::Qml Qt6::DBus Qt6::Sql)
find_package(KPim6Akonadi QUIET)
find_package(KPim6AkonadiMime QUIET)
if(KPim6Akonadi_FOUND AND KPim6AkonadiMime_FOUND)
  # KMime: part of KDE Frameworks from 2026 (KF6Mime, messages in std::shared_ptr), KPim6Mime before (QSharedPointer)
  find_package(KF6Mime QUIET)
  if(KF6Mime_FOUND)
    set(KMIME KF6::Mime)
  else()
    find_package(KPim6Mime REQUIRED)
    set(KMIME KPim6::Mime)
  endif()
  list(APPEND SOURCES kmailinbox.cpp kmailinbox.h)
  list(APPEND LIBS KPim6::AkonadiCore KPim6::AkonadiMime ${KMIME})
else()
  message(STATUS "No Akonadi: the KMail source is left out (Thunderbird only)")
endif()
qt_policy(SET QTP0001 NEW)
qt_add_qml_module(etesianmail URI org.etesian.mail VERSION 1.0 PLUGIN_TARGET etesianmail NO_GENERATE_PLUGIN_SOURCE
  CLASS_NAME EtesianMailPlugin SOURCES ${SOURCES} OUTPUT_DIRECTORY org/etesian/mail)
target_link_libraries(etesianmail PRIVATE ${LIBS})
if(KF6Mime_FOUND)
  target_compile_definitions(etesianmail PRIVATE KMIME_STD_SHARED_PTR)
endif()
# test/thunderbird.sh: the Thunderbird reader against a made-up profile (not installed)
add_executable(etesian-mail-thunderbird-test test/thunderbird.cpp thunderbirdinbox.cpp thunderbirdinbox.h thunderbirdsearch.cpp thunderbirdsearch.h)
find_package(Qt6 REQUIRED COMPONENTS Test)
target_link_libraries(etesian-mail-thunderbird-test PRIVATE Qt6::Core Qt6::Qml Qt6::Sql Qt6::Test)
